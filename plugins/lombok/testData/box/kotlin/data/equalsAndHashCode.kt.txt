  assertNotEquals<LocalClass>(illegal = LocalClass(x = 7), actual = LocalClass(x = 8))
  assertEquals<Color>(expected = Color.RED, actual = Color.RED)
  assertNotEquals<Color>(illegal = Color.RED, actual = Color.GREEN)
  assertEquals<Int>(expected = Color.RED.hashCode(), actual = Color.RED.hashCode())
  return "OK"
