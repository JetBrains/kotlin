package test

public open class AnnotatedField {
    public constructor AnnotatedField()
    @test.AnnotatedField.Anno(value = "member") public final val y: kotlin.Int

    @kotlin.annotation.Retention(value = ...) public final annotation class Anno : kotlin.Annotation {
        public final val value: kotlin.String
            public final fun `<get-value>`(): kotlin.String
    }

    // Static members
    @test.AnnotatedField.Anno(value = "static") public const final val x: kotlin.Int
}
